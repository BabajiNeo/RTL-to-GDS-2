# RTL Script to run Basic Synthesis Flow in Genus
set_db init_lib_search_path /home/vlsi12/Music/sclpdk/scl180/stdcell/fs120/6M1L/liberty/lib_flow_ss/

# Read library and design files
read_library tsl18fs120_scl_ss.lib
read_hdl -sv ripple_carry.v
elaborate  

# Read timing constraints  
read_sdc top.sdc  

# Configure DFT settings  
set_db dft_scan_style muxed_scan  
set_db dft_prefix "dft_"  
define_dft shift_enable -name SE -active high -create_port SE  
check_dft_rules  

# Synthesis steps  
set_db syn_generic_effort medium  
syn_generic  
set_db syn_map_effort medium  
syn_map  
set_db syn_opt_effort medium  
syn_opt  

# DFT insertion  
check_dft_rules  
set_db design:ripple_carry .dft_min_number_of_scan_chains 1  
define_dft scan_chain -name top_chain -sdi scan_in -sdo scan_out -create_ports  
connect_scan_chains -auto_create_chains -preview  
connect_scan_chains -auto_create_chains  
syn_opt -incr  

# Reports  
report_scan_chains  
write_dft_atpg -library ../lib/slow.v  
write_hdl > ripple_carry_netlist_dft.v  
write_sdc > ripple_carry.sdc  
write_sdf -edges check_edge -timescale ns -recrem split > delays_dft.sdf  
write_scandef > ripple_carry_scanDEF.scandef  

# Timing, Area, and Power Reports  
report_timing > ripple_carry.rep  
report_area > ripple_carry.rep  
report_gates > ripple_carry.rep  
report_power > ripple_carry.rep  
report_timing_summary > ripple_carry.rep  

# GUI (optional)  
gui_show
